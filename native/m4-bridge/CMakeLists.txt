cmake_minimum_required(VERSION 4.2)
project(m4_bridge_harness LANGUAGES C)
if(NOT WIN32)
  message(FATAL_ERROR "This harness requires Windows")
endif()
add_executable(m4_bridge_harness harness.c)
target_compile_definitions(m4_bridge_harness PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(m4_bridge_harness PRIVATE /W4 /WX /guard:cf)
target_link_options(m4_bridge_harness PRIVATE /guard:cf)
target_link_libraries(m4_bridge_harness PRIVATE advapi32 bcrypt)
enable_testing()
add_test(NAME windows_pipe_watchdog COMMAND m4_bridge_harness)
set_tests_properties(windows_pipe_watchdog PROPERTIES TIMEOUT 20)
