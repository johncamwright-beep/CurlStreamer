cmake_minimum_required(VERSION 4.2)
project(m4_ip_camera LANGUAGES C)
set(M4_AV_DEPS "" CACHE PATH "Matching pinned OBS FFmpeg development dependencies")
if(NOT EXISTS "${M4_AV_DEPS}/include/libavformat/avformat.h")
  message(FATAL_ERROR "M4_AV_DEPS must contain matching FFmpeg headers and libraries")
endif()
add_executable(m4_ip_camera camera.c)
target_include_directories(m4_ip_camera SYSTEM PRIVATE "${M4_AV_DEPS}/include")
target_link_directories(m4_ip_camera PRIVATE "${M4_AV_DEPS}/lib")
target_link_libraries(m4_ip_camera PRIVATE avformat avcodec avutil swscale swresample jansson ws2_32)
target_compile_definitions(m4_ip_camera PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(m4_ip_camera PRIVATE /W4 /WX /guard:cf)
target_link_options(m4_ip_camera PRIVATE /guard:cf)
option(M4_IP_CAMERA_FIXTURE "Build synthetic encoder for actual RTSP receiver validation" ON)
if(M4_IP_CAMERA_FIXTURE)
  add_executable(m4_ip_camera_fixture fixture.c)
  target_include_directories(m4_ip_camera_fixture PRIVATE "${M4_AV_DEPS}/include")
  target_link_directories(m4_ip_camera_fixture PRIVATE "${M4_AV_DEPS}/lib")
  target_link_libraries(m4_ip_camera_fixture PRIVATE avcodec avutil)
  target_compile_definitions(m4_ip_camera_fixture PRIVATE WIN32_LEAN_AND_MEAN)
endif()
