cmake_minimum_required(VERSION 4.2)
project(m4_log_guard LANGUAGES C)
find_package(libobs 32.2.2 EXACT CONFIG REQUIRED)
add_library(m4_log_guard STATIC m4-log-guard.c)
target_include_directories(m4_log_guard PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(m4_log_guard PUBLIC OBS::libobs)
target_compile_definitions(m4_log_guard PRIVATE WIN32_LEAN_AND_MEAN)
target_compile_options(m4_log_guard PRIVATE /W4 /WX /guard:cf)
add_executable(m4_log_guard_test test.c)
target_link_libraries(m4_log_guard_test PRIVATE m4_log_guard)
target_compile_definitions(m4_log_guard_test PRIVATE WIN32_LEAN_AND_MEAN)
target_compile_options(m4_log_guard_test PRIVATE /W4 /WX /guard:cf)
target_link_options(m4_log_guard_test PRIVATE /guard:cf)
set(M4_RUNTIME_BIN "" CACHE PATH "Pinned OBS runtime directory")
enable_testing()
add_test(NAME log_guard_real_obs_callback COMMAND m4_log_guard_test)
set_tests_properties(log_guard_real_obs_callback PROPERTIES TIMEOUT 15 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${M4_RUNTIME_BIN}")
