cmake_minimum_required(VERSION 4.2)
project(curlstreamer_m4_memory LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This validation slice supports Windows only")
endif()
option(M4_OBS_TEST_API "Expose synthetic-only validation controls; never distribute" OFF)
option(M4_OBS_LOOPBACK_TEST "Allow fixed loopback RTMP destination in test API only; never distribute" OFF)
option(M4_OBS_PRODUCTION_ADMISSION "Admit strict YouTube RTMPS targets over authenticated IPC; controlled rehearsal artifact only" OFF)
if(M4_OBS_PRODUCTION_ADMISSION AND (M4_OBS_TEST_API OR M4_OBS_LOOPBACK_TEST))
  message(FATAL_ERROR "Production admission and synthetic controls must be separate artifacts")
endif()
if(M4_OBS_LOOPBACK_TEST AND NOT M4_OBS_TEST_API)
  message(FATAL_ERROR "Loopback validation requires M4_OBS_TEST_API")
endif()
set(OBS_SDK_ROOT "" CACHE PATH "Pinned OBS 32.2.2 development SDK")
if(NOT EXISTS "${OBS_SDK_ROOT}/include/obs.h")
  message(FATAL_ERROR "Set OBS_SDK_ROOT to the pinned OBS 32.2.2 SDK")
endif()
add_library(curlstreamer-m4-memory MODULE memory-service.cpp)
target_compile_features(curlstreamer-m4-memory PRIVATE cxx_std_17)
target_include_directories(curlstreamer-m4-memory PRIVATE "${OBS_SDK_ROOT}/include")
target_link_libraries(curlstreamer-m4-memory PRIVATE "${OBS_SDK_ROOT}/lib/obs.lib" advapi32)
target_compile_definitions(curlstreamer-m4-memory PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX HAVE_OBSCONFIG_H)
if(M4_OBS_TEST_API)
  target_compile_definitions(curlstreamer-m4-memory PRIVATE M4_OBS_TEST_API=1)
endif()
if(M4_OBS_LOOPBACK_TEST)
  target_compile_definitions(curlstreamer-m4-memory PRIVATE M4_OBS_LOOPBACK_TEST=1)
endif()
if(M4_OBS_PRODUCTION_ADMISSION)
  target_compile_definitions(curlstreamer-m4-memory PRIVATE M4_OBS_PRODUCTION_ADMISSION=1)
endif()
target_compile_options(curlstreamer-m4-memory PRIVATE /W4 /WX /EHsc)
