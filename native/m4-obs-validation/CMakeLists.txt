cmake_minimum_required(VERSION 4.2)
project(m4_obs_validation LANGUAGES C)
find_package(libobs 32.2.2 EXACT CONFIG REQUIRED)
add_executable(m4_obs_validation host.c ipc-launch.c)
target_link_libraries(m4_obs_validation PRIVATE OBS::libobs bcrypt advapi32)
add_executable(m4_ipc_controller ipc-controller.c)
target_compile_definitions(m4_ipc_controller PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(m4_ipc_controller PRIVATE /W4 /WX /guard:cf)
target_link_options(m4_ipc_controller PRIVATE /guard:cf)
target_compile_definitions(m4_obs_validation PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(m4_obs_validation PRIVATE /W4 /WX /guard:cf)
target_link_options(m4_obs_validation PRIVATE /guard:cf)
set(M4_RUNTIME_BIN "" CACHE PATH "Pinned OBS runtime bin directory")
set(M4_TEST_PLUGIN "" CACHE FILEPATH "Test-enabled service DLL")
set(M4_DEFAULT_PLUGIN "" CACHE FILEPATH "Default service DLL")
set(M4_PRODUCTION_PLUGIN "" CACHE FILEPATH "Explicit production-admission service DLL; custom local sinks only")
set(M4_NODE_EXECUTABLE "" CACHE FILEPATH "Node runtime for application transport proof")
set(M4_NODE_SCRIPT "" CACHE FILEPATH "Node bootstrap proof script")
set(M4_STUDIO_SCRIPT "" CACHE FILEPATH "Prebuilt Studio bootstrap proof script")
if(M4_RUNTIME_BIN AND M4_TEST_PLUGIN AND M4_DEFAULT_PLUGIN)
  enable_testing()
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/test" "${CMAKE_BINARY_DIR}/evidence/default")
  add_test(NAME service_watchdog COMMAND m4_obs_validation "${M4_TEST_PLUGIN}" "${M4_RUNTIME_BIN}")
  add_test(NAME default_closed COMMAND m4_obs_validation "${M4_DEFAULT_PLUGIN}" "${M4_RUNTIME_BIN}" --production)
  set_tests_properties(service_watchdog PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/test" TIMEOUT 30)
  set_tests_properties(default_closed PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/default" TIMEOUT 15)
  set_tests_properties(service_watchdog default_closed PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${M4_RUNTIME_BIN}")
  foreach(scenario RANGE 0 11)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/ipc-${scenario}")
    if(scenario EQUAL 9)
      set(ipc_plugin "${M4_DEFAULT_PLUGIN}")
    else()
      set(ipc_plugin "${M4_TEST_PLUGIN}")
    endif()
    add_test(NAME "ipc_${scenario}" COMMAND m4_obs_validation "${ipc_plugin}" "${M4_RUNTIME_BIN}" --ipc "${scenario}")
    set_tests_properties("ipc_${scenario}" PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/ipc-${scenario}" TIMEOUT 20 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${M4_RUNTIME_BIN}")
  endforeach()
  if(M4_PRODUCTION_PLUGIN)
    foreach(scenario RANGE 12 15)
      file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/ipc-${scenario}")
      add_test(NAME "ipc_${scenario}" COMMAND m4_obs_validation "${M4_PRODUCTION_PLUGIN}" "${M4_RUNTIME_BIN}" --ipc "${scenario}")
      set_tests_properties("ipc_${scenario}" PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/ipc-${scenario}" TIMEOUT 20 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${M4_RUNTIME_BIN}")
    endforeach()
  endif()
  if(M4_NODE_EXECUTABLE AND M4_STUDIO_SCRIPT)
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/studio-bootstrap")
    add_test(NAME studio_bootstrap_default COMMAND m4_obs_validation "${M4_DEFAULT_PLUGIN}" "${M4_RUNTIME_BIN}" --studio-bootstrap)
    set_tests_properties(studio_bootstrap_default PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/studio-bootstrap" TIMEOUT 20 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${M4_RUNTIME_BIN}" ENVIRONMENT "M4_NODE_EXECUTABLE=${M4_NODE_EXECUTABLE};M4_STUDIO_SCRIPT=${M4_STUDIO_SCRIPT};NODE_OPTIONS=--no-warnings;NODE_PATH=C:/m4-inherited-path-must-not-reach-child;M4_PARENT_ENV_CANARY=must-not-reach-node")
  endif()
  if(M4_NODE_EXECUTABLE AND M4_NODE_SCRIPT)
    foreach(scenario RANGE 0 4)
      file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/node-${scenario}")
      add_test(NAME "node_ipc_${scenario}" COMMAND m4_obs_validation "${M4_TEST_PLUGIN}" "${M4_RUNTIME_BIN}" --node-ipc "${scenario}")
      set_tests_properties("node_ipc_${scenario}" PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/evidence/node-${scenario}" TIMEOUT 20 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${M4_RUNTIME_BIN}" ENVIRONMENT "M4_NODE_EXECUTABLE=${M4_NODE_EXECUTABLE};M4_NODE_SCRIPT=${M4_NODE_SCRIPT};NODE_OPTIONS=--no-warnings;NODE_PATH=C:/m4-inherited-path-must-not-reach-child;M4_PARENT_ENV_CANARY=must-not-reach-node")
    endforeach()
  endif()
endif()
