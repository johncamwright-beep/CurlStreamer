cmake_minimum_required(VERSION 4.2)
project(m4_studio_media LANGUAGES C)
find_package(libobs 32.2.2 EXACT CONFIG REQUIRED)
add_library(m4_studio_media STATIC m4-studio-media.c ../m4-log-guard/m4-log-guard.c)
target_include_directories(m4_studio_media PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(m4_studio_media PUBLIC OBS::libobs advapi32)
target_compile_definitions(m4_studio_media PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(m4_studio_media PRIVATE /W4 /WX /guard:cf)
option(M4_MEDIA_BUILD_VALIDATION "Build isolated real recording validation" ${PROJECT_IS_TOP_LEVEL})
if(M4_MEDIA_BUILD_VALIDATION)
  add_executable(m4_studio_media_validation validation.c)
  target_link_libraries(m4_studio_media_validation PRIVATE m4_studio_media)
  target_compile_definitions(m4_studio_media_validation PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(m4_studio_media_validation PRIVATE /W4 /WX /guard:cf)
  target_link_options(m4_studio_media_validation PRIVATE /guard:cf)
  set(M4_RUNTIME_BIN "" CACHE PATH "Pinned OBS 32.2.2 runtime bin")
  set(M4_FFMPEG "" CACHE FILEPATH "Local ffmpeg for decode validation")
  if(M4_RUNTIME_BIN AND M4_FFMPEG)
    add_custom_command(TARGET m4_studio_media_validation POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${M4_RUNTIME_BIN}/obs-ffmpeg-mux.exe" "$<TARGET_FILE_DIR:m4_studio_media_validation>/obs-ffmpeg-mux.exe")
    enable_testing()
    add_test(NAME independent_mkv COMMAND "${CMAKE_COMMAND}" "-DHOST=$<TARGET_FILE:m4_studio_media_validation>" "-DRUNTIME=${M4_RUNTIME_BIN}" "-DFFMPEG=${M4_FFMPEG}" "-DEVIDENCE=${CMAKE_BINARY_DIR}/evidence" -P "${CMAKE_CURRENT_SOURCE_DIR}/validate.cmake")
    set_tests_properties(independent_mkv PROPERTIES TIMEOUT 35 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${M4_RUNTIME_BIN}")
    set(M4_NODE "" CACHE FILEPATH "Node runtime for public browser fixture")
    if(M4_NODE)
      add_test(NAME public_browser_program COMMAND "${M4_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/program-validation.mjs" "$<TARGET_FILE:m4_studio_media_validation>" "${M4_RUNTIME_BIN}" "${M4_FFMPEG}" "${CMAKE_BINARY_DIR}/evidence")
      set_tests_properties(public_browser_program PROPERTIES TIMEOUT 40 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${M4_RUNTIME_BIN}")
      add_test(NAME exact_origin_host_addresses COMMAND "${M4_NODE}" "${CMAKE_CURRENT_SOURCE_DIR}/address-validation.mjs" "$<TARGET_FILE:m4_studio_media_validation>" "${M4_RUNTIME_BIN}" "${CMAKE_BINARY_DIR}/evidence")
      set_tests_properties(exact_origin_host_addresses PROPERTIES TIMEOUT 30)
    endif()
  endif()
endif()
