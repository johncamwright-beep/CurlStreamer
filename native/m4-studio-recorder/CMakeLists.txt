cmake_minimum_required(VERSION 4.2)
project(m4_studio_recorder LANGUAGES C)
add_subdirectory(../m4-studio-media media)
add_executable(m4_studio_recorder host.c stream-bootstrap.c)
target_link_libraries(m4_studio_recorder PRIVATE m4_studio_media bcrypt)
target_compile_definitions(m4_studio_recorder PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(m4_studio_recorder PRIVATE /W4 /WX /guard:cf)
target_link_options(m4_studio_recorder PRIVATE /guard:cf)
set(M4_RUNTIME_BIN "" CACHE PATH "Pinned OBS 32.2.2 runtime bin")
if(NOT EXISTS "${M4_RUNTIME_BIN}/obs-ffmpeg-mux.exe")
  message(FATAL_ERROR "M4_RUNTIME_BIN must contain the pinned OBS mux helper")
endif()
add_custom_command(TARGET m4_studio_recorder POST_BUILD
  COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${M4_RUNTIME_BIN}/obs-ffmpeg-mux.exe" "$<TARGET_FILE_DIR:m4_studio_recorder>/obs-ffmpeg-mux.exe")
set(M4_STREAM_PLUGIN "" CACHE FILEPATH "Default-deny production memory service DLL for isolated integration validation")
set(M4_FFMPEG "" CACHE FILEPATH "FFmpeg for recording decode validation")
set(M4_NODE "" CACHE FILEPATH "Node for authenticated recorder integration validation")
if(M4_STREAM_PLUGIN AND M4_FFMPEG AND M4_NODE)
  enable_testing()
  add_test(NAME recorder_stream_independence COMMAND "${M4_NODE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/stream-validation.mjs" "$<TARGET_FILE:m4_studio_recorder>"
    "${M4_RUNTIME_BIN}" "${M4_STREAM_PLUGIN}" "${M4_FFMPEG}" "${CMAKE_BINARY_DIR}/evidence")
  set_tests_properties(recorder_stream_independence PROPERTIES TIMEOUT 100)
endif()
