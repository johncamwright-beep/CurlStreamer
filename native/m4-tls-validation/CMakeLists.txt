cmake_minimum_required(VERSION 4.2)
project(m4_tls_validation LANGUAGES C CXX)
set(OBS_SOURCE "" CACHE PATH "Verified OBS 32.2.2 source")
set(M4_CA_FILE "" CACHE FILEPATH "Isolated test CA; never global trust")
find_package(libobs 32.2.2 EXACT CONFIG REQUIRED)
find_package(MbedTLS CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(jansson CONFIG REQUIRED)
if(NOT EXISTS "${M4_CA_FILE}" OR NOT EXISTS "${OBS_SOURCE}/plugins/obs-outputs/librtmp/rtmp.c")
  message(FATAL_ERROR "Verified source and generated isolated CA required")
endif()
file(READ "${OBS_SOURCE}/plugins/obs-outputs/librtmp/rtmp.c" RTMP)
set(NEEDLE "    mbedtls_x509_crt_init(chain);")
set(REPLACEMENT "    mbedtls_x509_crt_init(chain);\n#ifdef M4_ISOLATED_TEST_CA\n    if (mbedtls_x509_crt_parse_file(chain, M4_ISOLATED_TEST_CA) != 0) goto error;\n    mbedtls_ssl_conf_ca_chain(&r->RTMP_TLS_ctx->conf, chain, NULL);\n    return;\n#endif")
string(FIND "${RTMP}" "${NEEDLE}" POS)
if(POS LESS 0)
 message(FATAL_ERROR "Pinned CA injection point missing")
endif()
string(REPLACE "${NEEDLE}" "${REPLACEMENT}" RTMP "${RTMP}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rtmp.c" "${RTMP}")
add_subdirectory("${OBS_SOURCE}/deps/libcaption" caption)
add_subdirectory("${OBS_SOURCE}/shared/happy-eyeballs" happy)
add_subdirectory("${OBS_SOURCE}/shared/opts-parser" opts)
add_subdirectory("${OBS_SOURCE}/shared/bpm" bpm)
set(O "${OBS_SOURCE}/plugins/obs-outputs")
add_library(m4-test-outputs MODULE "${CMAKE_CURRENT_BINARY_DIR}/rtmp.c")
foreach(F flv-mux.c flv-output.c librtmp/amf.c librtmp/cencode.c librtmp/hashswf.c librtmp/log.c librtmp/md5.c librtmp/parseurl.c mp4-mux.c mp4-output.c net-if.c null-output.c obs-outputs.c rtmp-av1.c rtmp-stream.c rtmp-windows.c)
 target_sources(m4-test-outputs PRIVATE "${O}/${F}")
endforeach()
target_include_directories(m4-test-outputs PRIVATE "${O}" "${O}/librtmp")
target_compile_definitions(m4-test-outputs PRIVATE USE_MBEDTLS CRYPTO "M4_ISOLATED_TEST_CA=\"${M4_CA_FILE}\"" _CRT_SECURE_NO_WARNINGS)
target_link_libraries(m4-test-outputs PRIVATE OBS::libobs OBS::happy-eyeballs OBS::opts-parser OBS::bpm MbedTLS::mbedtls ZLIB::ZLIB jansson::jansson OBS::w32-pthreads crypt32 iphlpapi winmm ws2_32)
target_link_options(m4-test-outputs PRIVATE /IGNORE:4098)
file(READ "../m4-obs-plugin/memory-service.cpp" SERVICE)
string(REPLACE "rtmp://127.0.0.1:19359/live2" "rtmps://localhost:19360/live2" SERVICE "${SERVICE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/memory-service.cpp" "${SERVICE}")
add_library(m4-tls-memory MODULE "${CMAKE_CURRENT_BINARY_DIR}/memory-service.cpp")
target_compile_features(m4-tls-memory PRIVATE cxx_std_17)
target_compile_definitions(m4-tls-memory PRIVATE M4_OBS_TEST_API M4_OBS_LOOPBACK_TEST WIN32_LEAN_AND_MEAN NOMINMAX HAVE_OBSCONFIG_H)
target_link_libraries(m4-tls-memory PRIVATE OBS::libobs advapi32)
add_executable(m4_tls_validation host.c ../m4-log-guard/m4-log-guard.c)
target_link_libraries(m4_tls_validation PRIVATE OBS::libobs)
target_compile_definitions(m4_tls_validation PRIVATE WIN32_LEAN_AND_MEAN)
target_compile_options(m4_tls_validation PRIVATE /W4 /WX)
target_include_directories(bpm PRIVATE "${OBS_SOURCE}/libobs")
target_include_directories(m4-test-outputs PRIVATE "${OBS_SOURCE}/libobs")
target_sources(m4-test-outputs PRIVATE "${OBS_SOURCE}/libobs/obs-hevc.c")
target_compile_definitions(m4-test-outputs PRIVATE UNICODE _UNICODE)
target_sources(m4-test-outputs PRIVATE "${O}/rtmp-hevc.c")
target_compile_definitions(m4-test-outputs PRIVATE ENABLE_HEVC)
